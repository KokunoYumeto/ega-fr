\begin{remarks}[19.7.2]
\label{IV.19.7.2}
\begin{enumerate}
  \item[(i)] Lorsque l'on suppose que $\operatorname{gr}_{\mathfrak K}^\bullet(A)$ est un $(A/\mathfrak K)$-module \emph{plat}, l'hypothèse que la suite $f$ est $(A/\mathfrak K)$-régulière entraîne qu'elle est aussi $\operatorname{gr}_{\mathfrak K}^\bullet(A)$-régulière (\textbf{0}, 15.1.14). On notera que ce sera le cas lorsque $A$ \emph{et} $A/\mathfrak K$ \emph{sont des anneaux réguliers} (\textbf{0}, 17.3.6): en effet, $\mathfrak K$ est alors un idéal régulier de $A$ (19.1.2), donc quasi-régulier, et en localisant aux idéaux maximaux de $A$ contenant $\mathfrak K$ et utilisant (\textbf{0}, 15.1.7), on voit que $\operatorname{gr}_{\mathfrak K}^\bullet(A)$ est un $(A/\mathfrak K)$-module \emph{projectif}.
  \item[(ii)] Il serait intéressant de pouvoir prouver la dernière conclusion de (19.7.1) en supposant seulement qu'il existe une $A$-algèbre $B$ qui est un anneau noethérien, pour lequel $\mathfrak JB$ est contenu dans le radical de $B$, et que $M$ est un $B$-\emph{module} de type fini.
\end{enumerate}
\end{remarks}

\begin{corollary}[19.7.3]
\label{IV.19.7.3}
Soient $A$ un anneau local noethérien, $\mathfrak m$ son idéal maximal, $\mathfrak K$ un idéal de $A$ tel que l'anneau $A/\mathfrak K$ soit régulier et de dimension $n$, $M$ un $A$-module de type fini. Les conditions suivantes sont équivalentes:
\begin{enumerate}
  \item[\textup{a)}] $\operatorname{gr}_{\mathfrak K}^\bullet(M)$ est un $(A/\mathfrak K)$-module plat.
  \item[\textup{b)}] Si $P(T)$ et $Q(T)$ sont les séries de Poincaré des $(A/\mathfrak m)$-modules gradués $\operatorname{gr}_{\mathfrak m}^\bullet(M)$ et $\operatorname{gr}_{\mathfrak K}^\bullet(M)\otimes_{A/\mathfrak K}(A/\mathfrak m)$, on a
\end{enumerate}
\begin{equation}
  Q(T)=P(T)(1\setmin T)^n.
  \tag{19.7.3.1}\label{IV.19.7.3.1}
\end{equation}
\end{corollary}

Soit $\mathbf f=(f_1,\ldots,f_n)$ une suite d'éléments de $A$ tels que les images des $f_i$ dans $A/\mathfrak K$ forment un système régulier de paramètres de $A/\mathfrak K$ (\textbf{0}, 17.1.6), de sorte que $\mathbf f$ est une suite $(A/\mathfrak K)$-régulière; en outre, si $\mathfrak J$ est l'idéal engendré par $\mathbf f$, on a $\mathfrak J+\mathfrak K=\mathfrak m$, et comme $A/\mathfrak m$ est un corps, $\operatorname{gr}_{\mathfrak m}^\bullet(M)$ est un $(A/\mathfrak m)$-module plat. Comme $A$ est noethérien et $M$ un $A$-module de type fini, on peut appliquer l'équivalence de \emph{a)} et \emph{b)} dans (19.7.1). En outre, comme $A/\mathfrak m$ est un corps, et que $\psi_M$ est surjectif, le fait que $\psi_M$ soit bijectif équivaut à dire que pour tout $h\geq0$, $\operatorname{gr}_{\mathfrak m}^h(M)$ et le sous-module des éléments de degré $h$ de $(\operatorname{gr}_{\mathfrak K}^\bullet(M)\otimes_{A/\mathfrak K}(A/\mathfrak m))[T_1,\ldots,T_n]$ ont même rang sur $A/\mathfrak m$; mais pour le second de ces modules le rang en question est évidemment égal à $\sum_{r=0}^h\binom{r+n\setmin 1}{n\setmin 1}\operatorname{rg}(\operatorname{gr}_{\mathfrak K}^{h-r}(M)\otimes_{A/\mathfrak K}(A/\mathfrak m))$. Comme $(1\setmin T)^{-n}=\sum_{r=0}^{\infty}\binom{r+n\setmin 1}{n\setmin 1}T^r$, cela prouve que la relation (19.7.3.1) est nécessaire et suffisante pour que $\psi_M$ soit bijectif.

\begin{corollary}[19.7.4]
\label{IV.19.7.4}
Soient $X$ un préschéma localement noethérien, $Y$ un sous-préschéma fermé de $X$, régulier et connexe, $\sh F$ un $\sh O_X$-Module cohérent normalement plat \textup{(6.10.1)} le long de $Y$. Alors il existe une série formelle $R\in\mathbb{Q}[[T]]$ \textup{(indépendante de $x\in Y$)} telle que, pour tout $x\in Y$, la série de Poincaré de $\operatorname{gr}_{\mathfrak m_x}^\bullet(\sh F_x)$ soit donnée par la formule
\begin{equation}
  P_x(\sh F)(T)=R(T)(1\setmin T)^{-n}\qquad\text{où }n=\dim(\sh O_{Y,x}).
  \tag{19.7.4.1}\label{IV.19.7.4.1}
\end{equation}
\end{corollary}
