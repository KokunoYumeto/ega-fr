\subsection{Préschémas algébriques géométriquement réduits.}
\label{subsection:IV.2.4.6-fr}

\begin{proposition}[4.6.1]
\label{IV.2.4.6.1-fr}
Soient $k$ un corps, $X$ un $k$-préschéma, $\Omega$ une extension parfaite de
$k$. Les conditions suivantes sont équivalentes~:
\begin{enumerate}
\item[\textup{a)}] Pour tout $k$-préschéma réduit $S$, $X\times_k S$ est réduit.
\item[\textup{b)}] Pour toute extension $K$ de $k$, $X\otimes_k K$ est réduit.
\item[\textup{c)}] $X\otimes_k\Omega$ est réduit.
\item[\textup{d)}] Pour toute extension radicielle finie $k'$ de $k$, $X\otimes_k k'$ est
réduit.
\item[\textup{e)}] $X$ est réduit, et pour toute composante irréductible $X_\alpha$ de
$X$, de point générique $x_\alpha$, $\boldsymbol{k}(x_\alpha)$ est une extension séparable de
$k$.
\end{enumerate}
\end{proposition}

\begin{proof}
Il est trivial que \emph{a)} $\Rightarrow$ \emph{b)} $\Rightarrow$ \emph{c)}~; comme $k'$ peut être
considérée comme sous-extension de $\Omega$, on a vu (4.4.1) que \emph{c)} implique
\emph{d)}. Montrons que \emph{d)} implique \emph{e)}. En prenant $k'=k$, on voit d'abord que $X$
est réduit~; pour prouver que $\boldsymbol{k}(x_\alpha)$ est séparable sur $k$, il suffit de
montrer que $\boldsymbol{k}(x_\alpha)\otimes_k k'$ est un anneau réduit pour toute extension
finie radicielle $k'$ de $k$ (4.3.5). On peut se borner au cas où
$X=\operatorname{Spec}(A)$ est affine, $A$ étant une $k$-algèbre réduite~; alors
par hypothèse l'anneau $A\otimes_k k'$ est réduit (\textbf{I}, 5.1.4) et
$\boldsymbol{k}(x_\alpha)\otimes_k k'$ est un anneau de fractions de cet anneau, donc il est
réduit $(\mathbf{0}_{\mathrm I},1.2.8)$. Enfin, pour prouver que \emph{e)} entraîne \emph{a)}, on peut
se borner au cas où $X=\operatorname{Spec}(A)$, $S=\operatorname{Spec}(B)$ sont
affines, $A$ et $B$ étant des $k$-algèbres. Les $\boldsymbol{k}(x_\alpha)$ sont les corps des
fractions des anneaux quotients $A/\mathfrak p_\alpha$, où
$\mathfrak p_\alpha=\mathfrak j_{x_\alpha}$ sont les idéaux premiers minimaux de
$A$~; comme par hypothèse $\bigcap_\alpha\mathfrak p_\alpha=0$, $A$ est contenu
dans le produit $\prod_\alpha A/\mathfrak p_\alpha$, donc dans
$\prod_\alpha \boldsymbol{k}(x_\alpha)$~; on peut de même considérer $B$ comme une
sous-algèbre d'un produit $\prod_\beta L_\beta$ d'extensions de $k$. Il en
résulte que $A\otimes_k B$ s'identifie à une sous-algèbre de
$\left(\prod_\alpha \boldsymbol{k}(x_\alpha)\right)\otimes_k
\left(\prod_\beta L_\beta\right)$,
et ce produit tensoriel s'identifie lui-même à une sous-algèbre de
$\prod_{\alpha,\beta}\bigl(\boldsymbol{k}(x_\alpha)\otimes_k L_\beta\bigr)$
(Bourbaki, \emph{Alg.}, chap.~II, 3\textsuperscript{e} éd., \S~7,
n\textsuperscript{o}~7, prop.~15). Or par hypothèse les
$\boldsymbol{k}(x_\alpha)\otimes_k L_\beta$ sont réduits (4.3.5), donc il en est de même de
$A\otimes_k B$.
\end{proof}

\begin{definition}[4.6.2]
\label{IV.2.4.6.2-fr}
Si les conditions équivalentes \textup{a)} à \textup{e)} de la proposition \textup{(4.6.1)} sont
remplies, on dit que $X$ est \emph{séparable} (ou \emph{géométriquement réduit},
ou \emph{universellement réduit}) \textup{sur} $k$. On dit qu'un $k$-préschéma $X$ est
\emph{géométriquement intègre sur} $k$ si, pour toute extension $K$ de $k$,
$X\otimes_k K$ est intègre~; il revient au même (en vertu de \textup{(4.6.1)}) de dire
que $X$ est séparable et géométriquement irréductible sur $k$.
\end{definition}

$\operatorname{Spec}(A)$ est séparable sur $k$~; cela signifie donc que pour
toute extension $K$ de $k$, l'anneau $A_{(K)}=A\otimes_k K$ est réduit. On
