\subsection{Propriétés de transitivité.}

\begin{proposition}[6.6.1]
\label{IV.2.s6.6.1-fr}
Pour un préschéma localement noethérien $Z$, désignons par $\boldsymbol{P}(Z)$ l'une
quelconque des propriétés suivantes:
\begin{enumerate}
\item[\textup{a)}] $Z$ est un préschéma de Cohen-Macaulay.
\item[\textup{b)}] $Z$ vérifie $(S_n)$.
\item[\textup{c)}] $Z$ est régulier.
\item[\textup{d)}] $Z$ vérifie $(R_n)$.
\item[\textup{e)}] $Z$ est normal.
\item[\textup{f)}] $Z$ est réduit.
\end{enumerate}
Soient alors $X$, $Y$, $Z$ trois préschémas localement noethériens,
$f:X\to Y$, $g:Y\to Z$ deux morphismes.
\begin{enumerate}
\item[\textup{(i)}] Supposons que $f$ et $g$ soient plats et que pour tout
$y\in f(X)$ \textup{(resp.} tout $z\in g(Y)$), le préschéma $f^{-1}(y)$
\textup{(resp.} $g^{-1}(z)$) vérifie $\boldsymbol{P}$. Alors $h=g\circ f$ est plat et pour tout
$z\in h(X)$, $h^{-1}(z)$ vérifie $\boldsymbol{P}$.
\item[\textup{(ii)}] Supposons vérifiées les conditions suivantes: $f$ est
fidèlement plat, $h=g\circ f$ est plat, pour tout $y\in Y$, le
préschéma $f^{-1}(y)$ vérifie $\boldsymbol{P}$, et pour tout $z\in h(X)$ le
préschéma $h^{-1}(z)$ vérifie $\boldsymbol{P}$. Alors $g$ est plat, et pour tout
$z\in g(Y)$, $g^{-1}(z)$ vérifie $\boldsymbol{P}$.
\end{enumerate}
\end{proposition}

\begin{enumerate}
\item[(i)] On sait déjà (2.1.6) que $h$ est plat; d'autre part, pour
      tout $z\in Z$,
$f_z=f\otimes 1_{\mathbf{k}(z)}:h^{-1}(z)\longrightarrow g^{-1}(z)$ est plat (2.1.4), et pour tout $y\in g^{-1}(z)$, $f_z^{-1}(y)$ est
isomorphe à $f^{-1}(y)$ par transitivité des fibres (\textbf{I}, 3.6.4). La
conclusion résulte alors respectivement de (6.3.5, (ii)), (6.4.2),
(6.5.2, (ii)), (6.5.3, (ii)), (6.5.4, (ii)) et (3.3.5, (ii)).
\item[(ii)] On sait déjà que $g$ est plat (2.2.13); en outre, pour
      tout $z\in Z$, $f_z$ est fidèlement plat (2.2.13). La conclusion
      résulte alors respectivement de (6.3.5, (i)), (6.4.2),
(6.5.2, (i)), (6.5.3, (i)), (6.5.4, (ii)) et (2.1.13).
\end{enumerate}

\begin{remark}[6.6.2]
\label{IV.2.s6.6.2-fr}
Supposons $h$ plat, $f$ fidèlement plat, et supposons que pour tout
$z\in Z$, $h^{-1}(z)$ soit \emph{de coprofondeur} $\leqslant n$ (5.7.1); alors
il résulte du raisonnement de (6.6.1, (ii)) et de (6.3.2) que
$g^{-1}(z)$ est \emph{de coprofondeur} $\leqslant n$ pour tout $z\in g(Y)$ et
que pour tout $y\in Y$, $f^{-1}(y)$ est \emph{de coprofondeur} $\leqslant n$.
\end{remark}
